const actors=[];
function spriteFromPixels(pattern,paletteMap,scale=1){
  const h=pattern.length,w=Math.max(...pattern.map(r=>r.length));
  const tex=canvasTexture(w,h,(g)=>{
    g.clearRect(0,0,w,h);
    pattern.forEach((row,y)=>[...row].forEach((ch,x)=>{
      if(paletteMap[ch]){g.fillStyle=paletteMap[ch];g.fillRect(x,y,1,1)}
    }));
  });
  tex.magFilter=THREE.NearestFilter;tex.minFilter=THREE.NearestFilter;
  const mat=new THREE.SpriteMaterial({map:tex,transparent:true,alphaTest:.1,depthWrite:true});
  const s=new THREE.Sprite(mat);s.scale.set(w/h*scale,scale,1);return s;
}

let yaw=0,pitch=-.025;
const keys=new Set();
let yVel=0,onGround=true;
const SPEED=4.1,RUN=1.75;

function validPosition(x,z){
  const main = Math.abs(x)<=3.75 && z<=11 && z>=-58;
  const cross = Math.abs(z+16)<=3.75 && x>=-23 && x<=23;
  return main || cross;
}

window.addEventListener('keydown',e=>{
  if(!window.CARD_WORLD_INPUT?.entered && e.code!=='KeyF' && e.code!=='KeyR')return;
  keys.add(e.code);
  if(e.code==='Space'&&onGround){yVel=3.8;onGround=false;e.preventDefault();}
  if(e.code==='KeyR'){camera.position.set(0,1.72,10.5);yaw=0;pitch=-.025;e.preventDefault();}
  if(e.code==='KeyF'){
    if(!document.fullscreenElement) document.documentElement.requestFullscreen?.();
    else document.exitFullscreen?.();
    e.preventDefault();
  }
});
window.addEventListener('keyup',e=>keys.delete(e.code));

function resize(){
  const r=viewport.getBoundingClientRect();
  const cssW=Math.max(320,Math.floor(r.width)),cssH=Math.max(240,Math.floor(r.height));
  const scale=Math.min(.76,1280/cssW,760/cssH);
  renderer.setSize(Math.max(320,Math.floor(cssW*scale)),Math.max(240,Math.floor(cssH*scale)),false);
  camera.aspect=cssW/cssH;camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(viewport);resize();

const FRAME=1000/24;
let last=performance.now();
function animate(now){
  requestAnimationFrame(animate);
  if(document.hidden)return;
  const elapsed=now-last;if(elapsed<FRAME)return;
  const dt=Math.min(.06,elapsed/1000);last=now;
  if(inputState && (inputState.lookDX||inputState.lookDY)){
    yaw-=inputState.lookDX*.00235;
    pitch-=inputState.lookDY*.00205;
    pitch=Math.max(-1.05,Math.min(1.05,pitch));
    inputState.lookDX*=.18;
    inputState.lookDY*=.18;
    if(Math.abs(inputState.lookDX)<.02)inputState.lookDX=0;
    if(Math.abs(inputState.lookDY)<.02)inputState.lookDY=0;
  }
  if(keys.has('KeyQ'))yaw+=1.35*dt;
  if(keys.has('KeyE'))yaw-=1.35*dt;
  const entered=isEntered();
  const f=entered?(((keys.has('KeyW')||keys.has('ArrowUp')||inputState?.mouseForward)?1:0)-((keys.has('KeyS')||keys.has('ArrowDown')||inputState?.mouseReverse)?1:0)):0;
  const s=entered?((keys.has('KeyD')?1:0)-(keys.has('KeyA')?1:0)):0;
  if(f||s){
    const run=(keys.has('ShiftLeft')||keys.has('ShiftRight'))?RUN:1;
    const len=Math.hypot(f,s)||1,sp=SPEED*run*dt,ff=f/len,ss=s/len;
    const nx=camera.position.x+(-Math.sin(yaw)*ff+Math.cos(yaw)*ss)*sp;
    const nz=camera.position.z+(-Math.cos(yaw)*ff-Math.sin(yaw)*ss)*sp;
    if(validPosition(nx,nz)){camera.position.x=nx;camera.position.z=nz;}
  }
  yVel-=9.8*dt;camera.position.y+=yVel*dt;
  if(camera.position.y<=1.72){camera.position.y=1.72;yVel=0;onGround=true;}
  camera.rotation.y=yaw;camera.rotation.x=pitch;
  for(const a of actors)a.position.y=(a.userData.baseY||.98)+Math.sin(now*.0018+(a.userData.phase||0))*.022;
  renderer.render(scene,camera);
}
requestAnimationFrame(animate);
