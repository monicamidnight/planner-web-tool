function basic(color, opts={}){
  return new THREE.MeshBasicMaterial({
    color, fog:true, side:opts.side ?? THREE.FrontSide,
    transparent:!!opts.transparent, opacity:opts.opacity ?? 1,
    map:opts.map ?? null, alphaTest:opts.alphaTest ?? 0,
    depthWrite:opts.depthWrite ?? true, blending:opts.blending ?? THREE.NormalBlending
  });
}
function lambert(color, opts={}){
  return new THREE.MeshLambertMaterial({
    color, fog:true, side:opts.side ?? THREE.FrontSide,
    transparent:!!opts.transparent, opacity:opts.opacity ?? 1,
    map:opts.map ?? null, alphaTest:opts.alphaTest ?? 0,
    depthWrite:opts.depthWrite ?? true
  });
}

function canvasTexture(w,h,draw){
  const c=document.createElement('canvas'); c.width=w; c.height=h;
  const g=c.getContext('2d'); g.imageSmoothingEnabled=false; draw(g,w,h);
  const t=new THREE.CanvasTexture(c); t.magFilter=THREE.NearestFilter; t.minFilter=THREE.NearestFilter; t.colorSpace=THREE.SRGBColorSpace; return t;
}
function repeatTexture(tex,x,y){tex.wrapS=tex.wrapT=THREE.RepeatWrapping;tex.repeat.set(x,y);return tex}

function hexToRgb(hex){
  const raw = hex.startsWith('#') ? hex.slice(1) : hex;
  const n = parseInt(raw,16);
  return {r:(n>>16)&255,g:(n>>8)&255,b:n&255};
}
function rgbToCss({r,g,b},a=1){return a===1?`rgb(${r|0},${g|0},${b|0})`:`rgba(${r|0},${g|0},${b|0},${a})`;}
function shade(hex, amt){
  const c=hexToRgb(hex);
  return rgbToCss({r:Math.max(0,Math.min(255,c.r+amt)),g:Math.max(0,Math.min(255,c.g+amt)),b:Math.max(0,Math.min(255,c.b+amt))});
}
function drawPixelGrid(g,w,h,{base='#444',seam='#111',cell=4,bevel=20,alpha=1}){
  g.fillStyle=seam; g.fillRect(0,0,w,h);
  for(let y=0;y<h;y+=cell){
    for(let x=0;x<w;x+=cell){
      const inner=cell-1;
      const variant=((x/cell)+(y/cell))%3===0?10:((x/cell)*3+(y/cell)*5)%4===0?-10:0;
      g.fillStyle=shade(base,variant); g.globalAlpha=alpha; g.fillRect(x+1,y+1,inner,inner);
      g.fillStyle=shade(base,bevel); g.globalAlpha=.26*alpha; g.fillRect(x+1,y+1,inner,1); g.fillRect(x+1,y+1,1,inner);
      g.fillStyle=shade(base,-bevel); g.globalAlpha=.24*alpha; g.fillRect(x+1,y+inner,inner,1); g.fillRect(x+inner,y+1,1,inner);
    }
  }
  g.globalAlpha=1;
}
function brickTexture(base='#21131b',seam='#0f0c10',brick='#4e2336'){
  return canvasTexture(64,64,(g,w,h)=>{
    drawPixelGrid(g,w,h,{base,seam,cell:4,bevel:18});
    for(let y=0;y<h;y+=8){
      const off=(y/8)%2?8:0;
      for(let x=-off;x<w;x+=16){
        g.fillStyle=shade(brick,8); g.fillRect(x+2,y+2,11,4);
        g.fillStyle=shade(brick,28); g.fillRect(x+2,y+2,11,1);
        g.fillStyle=shade(brick,-22); g.fillRect(x+2,y+5,11,1);
      }
    }
    g.globalAlpha=.08; g.fillStyle='#fff'; g.fillRect(5,8,1,16); g.fillRect(26,30,1,9); g.globalAlpha=1;
  });
}

function signTexture(text,fg='#ff3f97',bg='rgba(8,8,10,.92)',vertical=false){
  const seed=Math.max(3,text.length);
  const cw=vertical?9:Math.max(15,seed*5+3), ch=vertical?Math.max(19,seed*6+3):11;
  return canvasTexture(cw,ch,(g,w,h)=>{
    g.fillStyle=bg;g.fillRect(0,0,w,h);
    // Abstract FAKE-BIT neon motif: no readable words.
    g.fillStyle=fg;
    if(vertical){
      for(let i=2;i<h-2;i+=5){g.fillRect(2,i,5,2);if((i/5)%2<1)g.fillRect(4,i-2,2,5)}
    }else{
      for(let x=2;x<w-3;x+=6){
        const y=2+((x/6)%3)*2;
        g.fillRect(x,y,4,2);
        if((x/6)%2<1)g.fillRect(x+2,y+2,2,4);
      }
      g.fillRect(2,h-3,w-4,1);
    }
    g.globalAlpha=.30;g.fillStyle='#fff';g.fillRect(2,1,Math.max(2,w-4),1);g.globalAlpha=1;
  });
}
function neonSign(text,colorHex,x,y,z,ry=0,vertical=false,scale=1,parent=world){
  const tex=signTexture(text,'#'+colorHex.toString(16).padStart(6,'0'),'rgba(5,5,7,.96)',vertical);
  const ratio=tex.image.width/tex.image.height, h=vertical?3.7:1.05*scale, w=h*ratio;
  const m=basic(0xffffff,{map:tex,transparent:true,opacity:.98,side:THREE.DoubleSide});
  const p=plane(w,h,m,x,y,z,0,ry,0,parent); glowSprite(x,y,z,colorHex,vertical?3.7:Math.max(2.6,w*.7),.24,parent); return p;
}
