
// LIVE CITY MOTION PASS
const cityMovers=[];
const P=(x,z,wait=0)=>({x,z,wait});

function addRouteMover(root,points,speed,{segment=0,t=0,kind='walker'}={}){
  const mover={root,points,speed,segment:Math.max(0,Math.min(points.length-1,segment|0)),t:Math.max(0,Math.min(.999,t)),wait:0,kind};
  const p0=points[mover.segment],p1=points[(mover.segment+1)%points.length];
  root.position.x=THREE.MathUtils.lerp(p0.x,p1.x,mover.t);
  root.position.z=THREE.MathUtils.lerp(p0.z,p1.z,mover.t);
  if(kind==='car'){
    const dx=p1.x-p0.x,dz=p1.z-p0.z;
    root.rotation.y=Math.atan2(-dx,-dz);
  }
  cityMovers.push(mover);
  return mover;
}

function makeWalker(variant,points,speed,segment=0,t=0){
  const root=new THREE.Group();
  world.add(root);
  robot(0,0,variant,root);
  return addRouteMover(root,points,speed,{segment,t,kind:'walker'});
}

function makeTrafficCar(color,points,speed,segment=0,t=0){
  const root=new THREE.Group();
  world.add(root);
  lowCar(0,0,0,color,root);
  return addRouteMover(root,points,speed,{segment,t,kind:'car'});
}

function angleStep(current,target,maxStep){
  let d=(target-current+Math.PI)%(Math.PI*2)-Math.PI;
  if(d<-Math.PI)d+=Math.PI*2;
  d=Math.max(-maxStep,Math.min(maxStep,d));
  return current+d;
}

function updateCityMotion(dt){
  for(const m of cityMovers){
    if(m.wait>0){m.wait=Math.max(0,m.wait-dt);continue;}
    let p0=m.points[m.segment],p1=m.points[(m.segment+1)%m.points.length];
    let dx=p1.x-p0.x,dz=p1.z-p0.z;
    let dist=Math.hypot(dx,dz);
    if(dist<.001){m.segment=(m.segment+1)%m.points.length;m.t=0;continue;}
    m.t+=m.speed*dt/dist;
    while(m.t>=1){
      m.t-=1;
      m.segment=(m.segment+1)%m.points.length;
      p0=m.points[m.segment];
      p1=m.points[(m.segment+1)%m.points.length];
      if(p0.wait)m.wait=Math.max(m.wait,p0.wait);
      dx=p1.x-p0.x;dz=p1.z-p0.z;dist=Math.max(.001,Math.hypot(dx,dz));
    }
    m.root.position.x=THREE.MathUtils.lerp(p0.x,p1.x,m.t);
    m.root.position.z=THREE.MathUtils.lerp(p0.z,p1.z,m.t);
    if(m.kind==='car'){
      const target=Math.atan2(-dx,-dz);
      m.root.rotation.y=angleStep(m.root.rotation.y,target,dt*4.8);
    }
  }
}

let __motionLast=performance.now();
const __baseAnimate=animate;
animate=function(now){
  const md=Math.min(.06,Math.max(0,(now-__motionLast)/1000));
  if(now-__motionLast>=FRAME){updateCityMotion(md);__motionLast=now;}
  __baseAnimate(now);
};
